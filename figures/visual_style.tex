\newcommand{\pPanel}[4]{%
  \node[pBadge] at (#3+.20,#4) {#1};
  \node[pTitle,anchor=west,inner sep=0pt] at (#3+.49,#4) {#2};
}
\newcommand{\pCandidate}[4]{%
  \node[pToken,fill=#2!16,draw=#2!40,line width=.5pt,text=#2!85!black] at (#3,#4) {#1};
}
\newcommand{\pComponent}[5]{%
  \node[draw=#2!40,fill=white,rounded corners=3pt,minimum width=.53cm,
    minimum height=.49cm,inner sep=0pt,pTitle,text=#2] at (#4,#5) {$#1$};
  \node[pSmall,align=center,inner sep=0pt] at (#4,#5-.43) {#3};
}
\newcommand{\pLock}[3]{%
  \draw[#3,line width=.65pt] (#1-.105,#2+.04) -- (#1-.105,#2+.15)
    arc[start angle=180,end angle=0,radius=.105] -- (#1+.105,#2+.04);
  \draw[#3,fill=#3!12,line width=.65pt,rounded corners=1pt]
    (#1-.17,#2-.16) rectangle (#1+.17,#2+.07);
  \fill[#3] (#1,#2-.035) circle (.025);
}

\newcommand{\pIcon}[5]{%
  \begin{scope}[shift={(#3,#4)},scale=#5,draw=pInk!85,line width=.65pt,
    line cap=round,line join=round]
    \def\pIconColor{#2}
    \pic {p/#1};
  \end{scope}
}
